================================================================================
NOTION NEST — TAB SETTINGS & TAB MENU EXPORT
For porting to another project / AI context
================================================================================

PURPOSE
-------
This pack is ONLY the Tab Settings (gear on tab bar) and per-tab Context Menu
(Rename, Edit icon, Copy link, Delete). It is NOT the BlockNote tab block
engine (visibility, Enter/Backspace, reparent) — that lives in notion-nest/tabs/.

TAB SETTINGS (block-level)
--------------------------
- UI: NotionTabsStylePopover — gear button on NotionPageTabsBar
- Portaled to document.body (#notion-tab-settings-host) so BlockNote remounts do not close it
- State: tabs/uiSession.ts (getTabsSettingsMenuOpen, setTabsSettingsMenuOpen)
- Persists on notionTabs block props via editor.updateBlock:
  tabStyle, tabShowUnderline, tabGap, tabHeight, tabCustomSelection,
  tabSelectionColor, tabAlignment
- Bar applies styles via notionTabShapes.ts (notionTabButtonClass, notionTabButtonStyle)

Options in panel:
  Shape | Tab Highlight (+ color) | Gap 0-32 | Align left/center/right | Height s/m/l

TAB MENU (per-tab)
------------------
- Open: double-click tab OR right-click tab (NotionPageTabsBar)
- State: setTabContextMenu in tabs/uiSession.ts
- Render: NotionTabOverlaysRoot (single mount on editor) -> NotionTabContextMenu
- Position: notionTabMenuAnchor.ts (above tab when space allows)

Menu actions:
  Rename     -> NotionTabRenameOverlay + commitTabRename (registry + panel props)
  Edit icon  -> setTabContextMenuMode("icon") -> NotionTabIconPicker
  Copy link  -> buildTabDeepLinkUrl (#tab:tabsId:panelId) + clipboard + toast
  Delete     -> NotionTabDeleteDialog -> deleteNotionTabPanelAtIndex (your tabs API)

REQUIRED MOUNT
--------------
  <NotionTabOverlaysRoot editor={editor} />  // next to BlockNote editor

REQUIRED DOM ON TAB BAR
-----------------------
  Tab wrap: data-tab-panel-id={panel.id}
  Tab button: role="tab"

EXTERNAL DEPS (copy or stub)
----------------------------
  components/ui/tabletemplates/utils/tabSettingsColorInputs  (NotionTabSettingsColorField)
  components/ui/tabletemplates/utils/tabBarIcons             (TabBarIcon, TAB_ICON_CUSTOM_KEY)
  components/ui/tabletemplates/utils/tabBarNormalize         (normalizeTabShape)
  tabs/registry.ts: readPanelRegistry, writePanelRegistry, updateRegistryPanelIcon
  tabs/activate.ts: deleteNotionTabPanelAtIndex
  notionUpload.uploadNotionFile (icon upload tab)
  lucide-react

INTEGRATION IN TAB BAR (see NotionPageTabsBar.tsx)
--------------------------------------------------
  readStyleProps + patchStyle + <NotionTabsStylePopover />
  openTabMenuForPanel + consumeTabDoublePointerDown + onContextMenu

function capturePanelPosition(anchor: HTMLElement | null): { top: number; left: number } | null {
    if (!anchor) return null;
    const r = anchor.getBoundingClientRect();
    if (r.width < 1 && r.height < 1) return null;
    return {
        top: Math.round(r.bottom + 4),
        left: Math.round(r.right - POPOVER_WIDTH_PX),
    };
}

let panelRoot: Root | null = null;
let panelHost: HTMLDivElement | null = null;

function ensurePanelHost(): Root {
    if (!panelHost) {
        panelHost = document.createElement('div');
        panelHost.id = HOST_ID;
        document.body.appendChild(panelHost);
        panelRoot = createRoot(panelHost);
    }
    return panelRoot!;
}

function destroyPanelHost() {
    panelRoot?.render(null);
    panelHost?.remove();
    panelHost = null;
    panelRoot = null;
}

type Props = {
    tabsBlockId: string;
    value: NotionTabsStyleProps;
    onChange: (patch: Partial<NotionTabsStyleProps>) => void;
};

export default function NotionTabsStylePopover({ tabsBlockId, value, onChange }: Props) {
    const anchorRef = useRef<HTMLDivElement>(null);
    const open = useSyncExternalStore(
        subscribeTabsUiSession,
        () => getTabsSettingsMenuOpen(tabsBlockId),
        () => getTabsSettingsMenuOpen(tabsBlockId),
    );

    const close = useCallback(() => {
        setTabsFieldMenuOpen(tabsBlockId, null);
        setTabsSettingsMenuOpen(tabsBlockId, false);
        if (!isAnyTabsSettingsOpen()) {
            destroyPanelHost();
        }
    }, [tabsBlockId]);

    const openPanel = useCallback(() => {
        const pos = capturePanelPosition(anchorRef.current) ?? getTabsPanelPosition(tabsBlockId);
        if (!pos) return;
        setTabsPanelPosition(tabsBlockId, pos);
        setTabsSettingsMenuOpen(tabsBlockId, true);
    }, [tabsBlockId]);

    const toggleOpen = useCallback(() => {
        if (open) {
            close();
            return;
        }
        openPanel();
    }, [open, close, openPanel]);

    useEffect(() => {
        if (!open) {
            if (!isAnyTabsSettingsOpen()) {
                destroyPanelHost();
            }
            return;
        }

        const pos = getTabsPanelPosition(tabsBlockId);
        if (!pos) return;

        const root = ensurePanelHost();
        root.render(
            <TabSettingsPanelContent
                position={pos}
                value={value}
                onChange={onChange}
                onRequestClose={close}
            />,
        );
    }, [open, tabsBlockId, value, onChange, close]);

    useEffect(() => {
        if (!open) return;

        const onKey = (e: KeyboardEvent) => {
            if (e.key === 'Escape') close();
        };
        window.addEventListener('keydown', onKey);

        const onOutside = (e: PointerEvent) => {
            if (e.button !== 0) return;
            const t = e.target;
            if (!(t instanceof Node)) return;
            if (anchorRef.current?.contains(t)) return;
            if (panelHost?.contains(t)) return;
            close();
        };

        const timer = window.setTimeout(() => {
            document.addEventListener('pointerdown', onOutside, true);
        }, 0);

        return () => {
            window.clearTimeout(timer);
            document.removeEventListener('pointerdown', onOutside, true);
            window.removeEventListener('keydown', onKey);
        };
    }, [open, close]);

    return (
        <div
            ref={anchorRef}
            className="notion-tabs-style-anchor relative shrink-0"
            data-notion-tabs-settings-anchor
        >
            <button
                type="button"
                className="notion-tabs-bar__icon-btn"
                title="Tab settings"
                aria-label="Tab settings"
                aria-expanded={open}
                data-no-row-nav
                onPointerDown={(e) => {
                    if (e.button !== 0) return;
                    stopBubble(e);
                    toggleOpen();
                }}
            >
                <Settings2 size={16} />
            </button>
        </div>
    );
}

import { Link2, Pencil, Smile, Trash2 } from 'lucide-react';

import { useEffect, useLayoutEffect, useRef, useState } from 'react';



import { TAB_ICON_CUSTOM_KEY, TabBarIcon } from '../../../components/ui/tabletemplates/utils/tabBarIcons';

import NotionTabIconPicker from './NotionTabIconPicker';

import { getTabContextMenuOpenedAt } from '../tabs';



import type { TabContextMenuPlacement } from '../tabs';

import {

    resolveFloatingMenuPosition,

    TAB_ICON_PICKER_HEIGHT,

    TAB_MENU_HEIGHT,

} from '../utils/notionTabMenuAnchor';



export type NotionTabContextMenuState = {

    panelId: string;

    top: number;

    left: number;

    mode: 'menu' | 'icon';

    placement?: TabContextMenuPlacement;

};



type Props = {

    state: NotionTabContextMenuState;

    tabLabel: string;

    tabIcon?: string;

    tabIconCustom?: string;

    onCopyLink: () => void | Promise<void>;

    onClose: () => void;

    onRename: () => void;

    onDelete: () => void;

    onOpenIconPicker: () => void;

    onIconChange: (iconKey: string, customIcon: string) => void;

};



const MENU_WIDTH = 200;

const ICON_PANEL_WIDTH = 360;



type MenuPosition = {

    top: number;

    left: number;

    placement: TabContextMenuPlacement;

};



export default function NotionTabContextMenu({

    state,

    tabLabel,

    tabIcon,

    tabIconCustom,

    onCopyLink,

    onClose,

    onRename,

    onDelete,

    onOpenIconPicker,

    onIconChange,

}: Props) {

    const rootRef = useRef<HTMLDivElement>(null);

    const [iconKeyDraft, setIconKeyDraft] = useState((tabIcon && String(tabIcon).trim()) || 'list');

    const [customDraft, setCustomDraft] = useState(tabIconCustom ?? '');

    const [position, setPosition] = useState<MenuPosition>(() => ({

        top: state.top,

        left: state.left,

        placement: state.placement ?? 'pointer',

    }));



    useEffect(() => {

        setIconKeyDraft((tabIcon && String(tabIcon).trim()) || 'list');

        setCustomDraft(tabIconCustom ?? '');

    }, [tabIcon, tabIconCustom, state.panelId, state.mode]);



    useLayoutEffect(() => {

        const panelWidth = state.mode === 'icon' ? ICON_PANEL_WIDTH : MENU_WIDTH;

        const menuHeight = state.mode === 'icon' ? TAB_ICON_PICKER_HEIGHT : TAB_MENU_HEIGHT;

        const anchor = getTabPanelAnchorRect(state.panelId);

        if (anchor) {

            const resolved = resolveFloatingMenuPosition(anchor, panelWidth, menuHeight);

            const next: MenuPosition = {

                top: resolved.top,

                left: resolved.left,

                placement: resolved.placement,

            };

            setPosition(next);

            return;

        }

        setPosition({

            top: state.top,

            left: Math.min(state.left, window.innerWidth - panelWidth - 8),

            placement: state.placement ?? 'pointer',

        });

    }, [state.panelId, state.mode, state.top, state.left, state.placement]);



    useEffect(() => {

        const onKey = (e: KeyboardEvent) => {

            if (e.key === 'Escape') onClose();

        };

        window.addEventListener('keydown', onKey);

        return () => window.removeEventListener('keydown', onKey);

    }, [onClose]);



    useEffect(() => {

        let armed = false;

        const armId = requestAnimationFrame(() => {

            armed = true;

        });

        const onOutside = (e: PointerEvent) => {

            if (!armed) return;

            if (performance.now() - getTabContextMenuOpenedAt() < 120) return;

            if (e.button !== 0) return;

            const t = e.target;

            if (!(t instanceof Node)) return;

            if (rootRef.current?.contains(t)) return;

            if (t instanceof Element && t.closest('[data-notion-tabs-bar]')) return;

            if (t instanceof Element && t.closest('[data-notion-tab-dialog]')) return;

            onClose();

        };

        document.addEventListener('pointerdown', onOutside, true);

        return () => {

            cancelAnimationFrame(armId);

            document.removeEventListener('pointerdown', onOutside, true);

        };

    }, [onClose]);



    const panelWidth = state.mode === 'icon' ? ICON_PANEL_WIDTH : MENU_WIDTH;

    const positionStyle =

        position.placement === 'above-tab'

            ? { top: position.top, left: position.left, transform: 'translateY(calc(-100% - 8px))' as const }

            : position.placement === 'below-tab'

              ? { top: position.top, left: position.left, transform: 'translateY(8px)' as const }

              : { top: position.top, left: position.left };



    const applyIcon = (key: string, custom: string) => {

        const trimmed = custom.trim();

        if (key === TAB_ICON_CUSTOM_KEY) {

            onIconChange(TAB_ICON_CUSTOM_KEY, trimmed);

            return;

        }

        onIconChange(key, '');

    };



    if (state.mode === 'icon') {

        return (

            <div

                ref={rootRef}

                className="notion-tab-context-menu notion-tab-context-menu--icon"

                role="dialog"

                aria-label="Edit tab icon"

                data-notion-tab-context-menu

                style={{ position: 'fixed', ...positionStyle, width: ICON_PANEL_WIDTH, zIndex: 10050 }}

                data-no-row-nav

                onPointerDown={(e) => e.stopPropagation()}

            >

                <NotionTabIconPicker

                    tabIcon={tabIcon}

                    tabIconCustom={tabIconCustom}

                    onIconChange={(key, custom) => {

                        setIconKeyDraft(key);

                        setCustomDraft(custom);

                        applyIcon(key, custom);

                    }}

                    onRemove={() => {

                        onIconChange('', '');

                        onClose();

                    }}

                    onClose={onClose}

                />

            </div>

        );

    }



    return (

        <div

            ref={rootRef}

            className="notion-tab-context-menu"

            role="menu"

            aria-label={`Tab options for ${tabLabel}`}

            data-notion-tab-context-menu

            style={{ position: 'fixed', ...positionStyle, width: MENU_WIDTH, zIndex: 10050 }}

            data-no-row-nav

            onPointerDown={(e) => e.stopPropagation()}

        >

            <button

                type="button"

                className="notion-tab-context-menu__item"

                role="menuitem"

                onMouseDown={(e) => {

                    e.preventDefault();

                    e.stopPropagation();

                    onRename();

                }}

                onClick={(e) => e.stopPropagation()}

            >

                <Pencil className="w-4 h-4 shrink-0" />

                Rename

            </button>

            <button

                type="button"

                className="notion-tab-context-menu__item"

                role="menuitem"

                onClick={(e) => {

                    e.stopPropagation();

                    onOpenIconPicker();

                }}

            >

                <Smile className="w-4 h-4 shrink-0" />

                <span className="flex-1 text-left">Edit icon</span>

                <TabBarIcon iconKey={tabIcon} customIcon={tabIconCustom} size={16} />

            </button>

            <button

                type="button"

                className="notion-tab-context-menu__item"

                role="menuitem"

                onClick={() => void onCopyLink()}

            >

                <Link2 className="w-4 h-4 shrink-0" />

                Copy link

            </button>

            <div className="notion-tab-context-menu__divider" />

            <button

                type="button"

                className="notion-tab-context-menu__item notion-tab-context-menu__item--danger"

                role="menuitem"

                onMouseDown={(e) => {

                    e.preventDefault();

                    e.stopPropagation();

                    onDelete();

                }}

            >

                <Trash2 className="w-4 h-4 shrink-0" />

}

export default function NotionTabOverlaysRoot({ editor }: { editor: NotionBlockNoteEditor }) {
    const { objectId, recordId } = useParams<{ objectId: string; recordId: string }>();
    const pagePath = useMemo(
        () => `/notion/${encodeURIComponent(String(objectId ?? ''))}/${encodeURIComponent(String(recordId ?? ''))}`,
        [objectId, recordId],
    );

    const overlayKey = useSyncExternalStore(
        subscribeTabsUiSession,
        buildGlobalOverlayKey,
        buildGlobalOverlayKey,
    );

    const lastOverlayKeyRef = useRef('');

    useEffect(() => {
        if (overlayKey === lastOverlayKeyRef.current) return;
        lastOverlayKeyRef.current = overlayKey;

        const nodes: ReactNode[] = [];
        const menu = getTabContextMenu();

        if (menu) {
            const tabs = editor.getBlock(menu.tabsBlockId);
            if (tabs?.type === 'notionTabs') {
                const panels = getTabPanels(tabs);
                const registryPanel = panels.find((p) => p.id === menu.panelId);
                if (registryPanel) {
                    const livePanel = resolveEditorBlock(editor, menu.panelId);
                    const panel = livePanel ?? registryPanel;
                    const panelIndex = panels.findIndex((p) => p.id === menu.panelId);
                    nodes.push(
                        <NotionTabContextMenu
                            key="menu"
                            state={{
                                panelId: menu.panelId,
                                top: menu.top,
                                left: menu.left,
                                mode: menu.mode,
                                placement: menu.placement,
                            }}
                            tabLabel={String(panel.props.label ?? 'Tab')}
                            tabIcon={String(panel.props.tabIcon ?? '')}
                            tabIconCustom={String(panel.props.tabIconCustom ?? '')}
                            onCopyLink={async () => {
                                const url = buildTabDeepLinkUrl(pagePath, menu.tabsBlockId, panel.id);
                                const ok = await copyTextToClipboard(url);
                                showTabLinkCopiedToast(
                                    ok
                                        ? 'Tab link copied'
                                        : 'Could not copy â€” copy URL from address bar after switching tabs',
                                );
                                setTabContextMenu(null);
                            }}
                            onClose={() => setTabContextMenu(null)}
                            onRename={() => {
                                const label = String(panel.props.label ?? 'Tab');
                                startTabRename(menu.tabsBlockId, panel.id, label);
                                setTabContextMenu(null);
                            }}
                            onDelete={() => {
                                setTabContextMenu(null);
                                if (panels.length <= 1) {
                                    showTabLinkCopiedToast('You must keep at least one tab.');
                                    return;
                                }
                                setTabDeleteConfirm({
                                    tabsBlockId: menu.tabsBlockId,
                                    panelId: panel.id,
                                    panelIndex,
                                    label: String(panel.props.label ?? `Tab ${panelIndex + 1}`),
                                });
                            }}
                            onOpenIconPicker={() => setTabContextMenuMode('icon')}
                            onIconChange={(iconKey, customIcon) => {
                                const custom =
                                    iconKey === TAB_ICON_CUSTOM_KEY ? customIcon : '';
                                const tabsLive = resolveEditorBlock(editor, menu.tabsBlockId);
                                if (tabsLive?.type === 'notionTabs') {
                                    const registry = updateRegistryPanelIcon(
                                        readPanelRegistry(tabsLive),
                                        panel.id,
                                        iconKey,
                                        custom,
                                    );
                                    editor.updateBlock(tabsLive, {
                                        props: writePanelRegistry(tabsLive, registry),
                                    });
                                }
                                const livePanel = resolveEditorBlock(editor, panel.id);
                                if (livePanel) {
                                    editor.updateBlock(livePanel, {
                                        props: {
                                            ...livePanel.props,
                                            tabIcon: iconKey,
                                            tabIconCustom: custom,
                                        },
                                    });
                                }
                            }}
                        />,
                    );
                }
            }
        }

        const deleteConfirm = getTabDeleteConfirm();
        if (deleteConfirm) {
            nodes.push(
                <NotionTabDeleteDialog
                    key="delete"
                    label={deleteConfirm.label}
                    onCancel={() => setTabDeleteConfirm(null)}
                    onConfirm={() => {

    const fallbackFontPx = tabHeight === 'small' ? 12 : tabHeight === 'large' ? 15 : 14;

        tabProps.tabAlignment === 'center' || tabProps.tabAlignment === 'right'

            ? tabProps.tabAlignment

        });

    }, [tabRename?.epoch, tabRename?.panelId]);



    if (!tabRename || !fieldBox) return null;

            focusedEpochRef.current = -1;

        }

            ref={inputRef}

                position: 'fixed',

export type TabMenuAnchor = {
    top: number;
    left: number;
    width: number;
    height: number;
    bottom: number;
};

export type TabMenuPlacement = 'above-tab' | 'below-tab';

export type FloatingMenuPosition = {
    top: number;
    left: number;
    placement: TabMenuPlacement;
};

export const TAB_MENU_HEIGHT = 220;
export const TAB_ICON_PICKER_HEIGHT = 400;

export function getTabRenameTargetElement(panelId: string): HTMLElement | null {
    const esc = typeof CSS !== 'undefined' && 'escape' in CSS ? CSS.escape(panelId) : panelId;
    const wrap = document.querySelector(`[data-tab-panel-id="${esc}"]`);
    if (!(wrap instanceof HTMLElement)) return null;
    const tab =
        wrap.querySelector<HTMLElement>('button[role="tab"]') ??
        wrap.querySelector<HTMLElement>('[role="tab"]') ??
        wrap.querySelector<HTMLElement>('button');
    return tab ?? wrap;
}

export function getTabPanelAnchorRect(panelId: string): TabMenuAnchor | null {
    const target = getTabRenameTargetElement(panelId);
    if (!target) return null;
    const rect = target.getBoundingClientRect();
    if (rect.width <= 0 && rect.height <= 0) return null;
    return {
        top: rect.top,
        left: rect.left,
        width: rect.width,
        height: rect.height,
        bottom: rect.bottom,
    };
}

export function resolveFloatingMenuPosition(
    anchor: TabMenuAnchor,
    menuWidth: number,
    menuHeight: number,
): FloatingMenuPosition {
    const margin = 8;
    const left = Math.max(margin, Math.min(anchor.left, window.innerWidth - menuWidth - margin));
    const spaceAbove = anchor.top - margin;
    const spaceBelow = window.innerHeight - anchor.bottom - margin;

    if (spaceAbove >= menuHeight + margin || spaceAbove >= spaceBelow) {
        return { top: anchor.top, left, placement: 'above-tab' };
    }
    return { top: anchor.bottom, left, placement: 'below-tab' };
}

export function menuPositionAboveTab(
    anchor: TabMenuAnchor,
    menuWidth: number,
    menuHeight = TAB_MENU_HEIGHT,
): FloatingMenuPosition {
    return resolveFloatingMenuPosition(anchor, menuWidth, menuHeight);
}

export function commitTabRename(
    editor: NotionBlockNoteEditor,
    draft: string,
    force = false,
): boolean {
    if (!force && isTabRenameGraceActive()) return false;
    const state = getTabRename();
    if (!state) return false;
    const next = draft.trim() || 'Tab';
    const tabs = resolveEditorBlock(editor, state.tabsBlockId);

        const registry = readPanelRegistry(tabs).map((e) =>
            e.id === state.panelId ? { ...e, label: next } : e,
        );
        editor.updateBlock(tabs, { props: writePanelRegistry(tabs, registry) });
    }
    const livePanel = resolveEditorBlock(editor, state.panelId);
    if (livePanel) {
        editor.updateBlock(livePanel, { props: { ...livePanel.props, label: next } });
    }
    setTabRename(null);
    return true;
}

import type { Block, BlockNoteEditor } from '@blocknote/core';
import { activateNotionTab, findBlockInTree, getTabPanels } from '../tabs';

const TAB_UUID =
    '([0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12})';

export function resolveEditorBlock(editor: BlockNoteEditor, blockId: string): Block | null {
    const direct = editor.getBlock(blockId);
    if (direct) return direct as Block;
    const found = findBlockInTree(editor.document as Block[], blockId);
    return found?.block ?? null;
}

/** Hash format: #tab:{tabsBlockId}:{panelId} (used for copy-link only; tab clicks do not update the URL). */
export function buildTabDeepLinkHash(tabsBlockId: string, panelId: string): string {
    return `tab:${tabsBlockId}:${panelId}`;
}

export function buildTabDeepLinkUrl(pagePath: string, tabsBlockId: string, panelId: string): string {
    const path = pagePath.startsWith('/') ? pagePath : `/${pagePath}`;
    return `${window.location.origin}${path}#${buildTabDeepLinkHash(tabsBlockId, panelId)}`;
}

export function parseTabDeepLinkHash(hash: string): { tabsBlockId: string; panelId: string } | null {
    const raw = hash.startsWith('#') ? hash.slice(1) : hash;
    const uuidPair = raw.match(new RegExp(`^tab:${TAB_UUID}:${TAB_UUID}$`, 'i'));
    if (uuidPair) return { tabsBlockId: uuidPair[1], panelId: uuidPair[2] };

    const loose = raw.match(/^tab:([^:]+):([^:]+)$/);
    if (loose) return { tabsBlockId: loose[1], panelId: loose[2] };

    if (raw.startsWith('tab-')) {
        const rest = raw.slice(4);
        const panelMatch = rest.match(
            /([0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12})$/i,
        );
        if (panelMatch) {
            const panelId = panelMatch[1];
            const tabsBlockId = rest.slice(0, rest.length - panelId.length - 1);
            if (tabsBlockId) return { tabsBlockId, panelId };
        }
    }
    return null;
}

/** Remove tab deep-link hash from the address bar (tab clicks should not change the URL). */
export function clearTabUrlHashIfPresent(): void {
    const hash = window.location.hash;
    if (!hash || !parseTabDeepLinkHash(hash)) return;
    const base = `${window.location.pathname}${window.location.search}`;
    window.history.replaceState(window.history.state, '', base);
}

function scrollTabsBlockIntoView(tabsBlockId: string): void {
    const esc =
        typeof CSS !== 'undefined' && typeof CSS.escape === 'function'
            ? CSS.escape(tabsBlockId)
            : tabsBlockId;
    const outer = document.querySelector(
        `.bn-block-outer[data-id="${esc}"].notion-tabs-outer`,
    ) as HTMLElement | null;
    outer?.scrollIntoView({ behavior: 'smooth', block: 'start' });
}

/**
 * Open the tab from `#tab:{tabsBlockId}:{panelId}` when present in the URL.
 * Returns true when the deep link was applied.
 */
export function applyTabDeepLinkFromHash(editor: BlockNoteEditor): boolean {
    const parsed = parseTabDeepLinkHash(window.location.hash);
    if (!parsed) return false;

    const tabs = resolveEditorBlock(editor, parsed.tabsBlockId);
    if (!tabs || tabs.type !== 'notionTabs') return false;

    const panels = getTabPanels(tabs);
    const index = panels.findIndex((p) => p.id === parsed.panelId);
    if (index < 0) return false;

    const ok = activateNotionTab(editor, parsed.tabsBlockId, index);
    if (ok) {
        requestAnimationFrame(() => scrollTabsBlockIntoView(parsed.tabsBlockId));
    }
    return ok;
}

/** Copy text with Clipboard API, falling back to execCommand for non-secure contexts. */
export async function copyTextToClipboard(text: string): Promise<boolean> {
    try {
        if (navigator.clipboard?.writeText) {
            await navigator.clipboard.writeText(text);
            return true;
        }
    } catch {
        /* fall through */
    }
    try {
        const ta = document.createElement('textarea');
        ta.value = text;
        ta.setAttribute('readonly', '');
        ta.style.position = 'fixed';
        ta.style.left = '-9999px';
        document.body.appendChild(ta);
        ta.select();
        const ok = document.execCommand('copy');
        ta.remove();
        return ok;
    } catch {
        return false;
    }
}

import { createRoot, type Root } from 'react-dom/client';
import type { ReactNode } from 'react';

const OVERLAY_HOST_ID = 'notion-tab-overlays-host';

let overlayRoot: Root | null = null;
let overlayHost: HTMLDivElement | null = null;

function ensureOverlayHost(): Root {
    if (!overlayHost) {
        overlayHost = document.createElement('div');
        overlayHost.id = OVERLAY_HOST_ID;
        overlayHost.style.position = 'fixed';
        overlayHost.style.left = '0';
        overlayHost.style.top = '0';
        overlayHost.style.width = '0';
        overlayHost.style.height = '0';
        overlayHost.style.overflow = 'visible';
        overlayHost.style.zIndex = '10050';
        overlayHost.style.pointerEvents = 'none';
        document.body.appendChild(overlayHost);
        overlayRoot = createRoot(overlayHost);
    }
    return overlayRoot!;
}

export function renderTabOverlays(node: ReactNode | null): void {
    if (!node) {
        if (!overlayHost) return;
        overlayRoot?.render(null);
        overlayHost.remove();
        overlayHost = null;
        overlayRoot = null;
        return;
    }
    ensureOverlayHost().render(node);
}
